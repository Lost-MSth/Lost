# 魔导士小奏的扶她地牢探险

魔導士カナデのふたなりダンジョン Quest  Mage Kanade’s Futanari Dungeon Quest

dieselmine-Int'l- ディーゼルマイン

## 类型

- 1.55 GB
- ACT、像素
- 架空世界
- 单线，加上一堆败北结局

## 评价

通关一遍不到两小时，收集了全部败北事件，总体收集了大部分的 H 事件，应该是漏了几个小差分（注意回想界面的红斜杠说明是空项目，未解锁项目是有虚影）。

总体而言不错，喜欢 futa 或者是 2D ACT 动作类游戏的可以试一试。

剧情倒也没什么好说的，进入地牢的女主小奏被变成了 futa，逃离地牢就是主要目标。每一层都有一个管理者，风格都不一样，然后还有个作为设施的主人的魅魔。逃出去就结束了，最后魅魔还有个自己下属群奸的事件。

游戏玩法倒是挺不错的，操作流畅，打击感、动画都挺好的，难度适中。用好闪避然后直接怼脸输出基本没有打不过的，当然不是无伤，而且我为了收集事件基本一路送过去的。

H 事件一般多，主要是血量归零的败北事件、固定事件、每个小怪或者 boss 的异常状态动画及事件。主要类型显然是 futa，另外以调教、榨精、紧身全包、催眠为主。CG 画风中等偏下，但是伪像素立绘动画中等偏上。部分 H 事件有配音，有音效。

## 破解

数据无加密，AssetStudio 可以看。但是存档有加密，是 `.save` 文件，位置在 `C:\Users\<你的用户名>\AppData\LocalLow\Dieselmine\Mage Kanades Futanari Dungeon Quest\Save` 里面。

注意到游戏使用了 `GameAssembly.dll`，逆向就困难多了，不过 `global-metadata.dat` 没有加密，挺好，直接 Il2CppDumper 开搞。

不想看解密逻辑，直接丢给 deepseek v4 pro。很多轮对话后我大概知道了这是一种 MessagePack 序列化格式，然后存档分为两个部分，头部和主体，是分别 AES‑256‑CBC 加密然后拼在一起的，结构就是 `[Header 块总长度 (int32 LE)] [Header 加密块] [Body 块总长度 (int32 LE)] [Body 加密块]`，当然密钥一样。

让 AI 写出解密脚本，甚至帮我翻译到了 JSON：

```python
import json
import secrets
import struct
import sys

import msgpack
from Crypto.Cipher import AES
from Crypto.Util.Padding import pad, unpad

# ========== 常量 ==========
ENCRYPT_KEY = b"UcdHfaxaCjHA5wFR2hcTU42N0zWH8B1m"
BLOCK_SIZE = 16
PASSWORD_CHARS = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"
PASSWORD_LENGTH = 16

# ========== 基础加解密 ==========


def decrypt_block(encrypted_block):
    off = 0
    iv_len = struct.unpack_from('<i', encrypted_block, off)[0]
    off += 4
    iv = encrypted_block[off:off + iv_len]
    off += iv_len
    cipher_len = struct.unpack_from('<i', encrypted_block, off)[0]
    off += 4
    ciphertext = encrypted_block[off:off + cipher_len]
    cipher = AES.new(ENCRYPT_KEY, AES.MODE_CBC, iv)
    return unpad(cipher.decrypt(ciphertext), BLOCK_SIZE)


def encrypt_block(plain):
    password = ''.join(secrets.choice(PASSWORD_CHARS)
                       for _ in range(PASSWORD_LENGTH))
    iv = password.encode('utf-8')
    cipher = AES.new(ENCRYPT_KEY, AES.MODE_CBC, iv)
    ciphertext = cipher.encrypt(pad(plain, BLOCK_SIZE))
    output = struct.pack('<i', len(iv))
    output += iv
    output += struct.pack('<i', len(ciphertext))
    output += ciphertext
    return output


# ========== 自定义布尔转换（游戏将 bool 存为 1 字节二进制） ==========
TRUE_BIN = b'\xc3'
FALSE_BIN = b'\xc2'


def bin_to_bool(val):
    """如果 val 是长度为 1 的 bytes，则转换为 bool，否则原样返回"""
    if isinstance(val, bytes) and len(val) == 1:
        return val == TRUE_BIN
    return val


def bool_to_bin(val):
    """将 Python bool 转换为对应的 1 字节二进制"""
    return TRUE_BIN if val else FALSE_BIN

# ========== 存档转 JSON ==========


def save_to_json(save_path, json_path):
    with open(save_path, 'rb') as f:
        data = f.read()
    pos = 0
    hdr_len = struct.unpack_from('<i', data, pos)[0]
    pos += 4
    hdr_enc = data[pos:pos + hdr_len]
    pos += hdr_len
    bdy_len = struct.unpack_from('<i', data, pos)[0]
    pos += 4
    bdy_enc = data[pos:pos + bdy_len]

    header = msgpack.unpackb(decrypt_block(
        hdr_enc), raw=False, strict_map_key=False)
    body = msgpack.unpackb(decrypt_block(
        bdy_enc), raw=False, strict_map_key=False)

    # 构建结构化数据，同时转换事件标志为 bool
    events_js = []
    for evt in body[3]:
        evt_id = evt[0].decode(
            'utf-8') if isinstance(evt[0], bytes) else evt[0]
        evt_flag = bin_to_bool(evt[1])   # 将二进制转为 True/False
        events_js.append([evt_id, evt_flag])

    save_obj = {
        "header": {
            "timestamp": header[0],
            "save_type": header[1].decode('utf-8') if isinstance(header[1], bytes) else header[1]
        },
        "body": {
            "location": body[0].decode('utf-8') if isinstance(body[0], bytes) else body[0],
            "player_pos": body[1],
            "unknown_int": body[2],
            "events": events_js
        }
    }

    with open(json_path, 'w', encoding='utf-8') as f:
        json.dump(save_obj, f, indent=2, ensure_ascii=False)
    print(f"已导出 JSON 到 {json_path}")

# ========== JSON 转存档 ==========


def json_to_save(json_path, save_path):
    with open(json_path, 'r', encoding='utf-8') as f:
        save_obj = json.load(f)

    # 事件标志恢复为二进制字节
    events_raw = []
    for evt in save_obj["body"]["events"]:
        events_raw.append([evt[0], bool_to_bin(evt[1])])

    header = [
        save_obj["header"]["timestamp"],
        save_obj["header"]["save_type"]
    ]
    body = [
        save_obj["body"]["location"],
        save_obj["body"]["player_pos"],
        save_obj["body"]["unknown_int"],
        events_raw
    ]

    # 打包时不对二进制数据做额外处理，直接使用默认选项
    header_bin = msgpack.packb(header, use_bin_type=True, strict_types=False)
    body_bin = msgpack.packb(body, use_bin_type=True, strict_types=False)

    hdr_enc = encrypt_block(header_bin)
    bdy_enc = encrypt_block(body_bin)

    with open(save_path, 'wb') as f:
        f.write(struct.pack('<i', len(hdr_enc)))
        f.write(hdr_enc)
        f.write(struct.pack('<i', len(bdy_enc)))
        f.write(bdy_enc)
    print(f"已生成存档 {save_path}")


# ========== 命令行接口 ==========
if __name__ == '__main__':
    if len(sys.argv) < 4:
        print("用法:")
        print("  decrypt <存档> <json>   - 解密到 JSON")
        print("  encrypt <json> <存档>   - 从 JSON 生成存档")
        sys.exit(1)
    mode = sys.argv[1]
    if mode == "decrypt":
        save_to_json(sys.argv[2], sys.argv[3])
    elif mode == "encrypt":
        json_to_save(sys.argv[2], sys.argv[3])
    else:
        print("未知模式，请使用 decrypt 或 encrypt")
```

这下随便存档全解锁了。
